## 题目简述

给出一个有理数 $c=\frac{a}{b}$，求 $c \bmod 19260817$ 的值。

这个值被定义为 $bx\equiv a\pmod{19260817}$ 的解。

输入格式
一共两行。

第一行，一个整数 $a$。
第二行，一个整数 $b$。

输出格式
一个整数，代表求余后的结果。如果无解，输出 `Angry!`。

说明/提示
对于所有数据，保证 $0\leq a \leq 10^{10001}$，$1 \leq b \leq 10^{10001}$，且 $a, b$ 不同时是 $19260817$ 的倍数。

#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD=19260817;
ll readbig(){
    string s;
    cin >> s;
    ll res = 0;
    for(char ch : s){
        res = (res * 10 + ch - '0') % MOD;
    }
    return res;
}

ll qpow(ll a,ll b,ll mod){
    ll res=1;
    while(b){
        if(b&1){
            res=res*a%mod;
           
        }
         a=a*a%mod;
         b>>=1;
    }
    return res;
}


int main(){
    ll a=readbig();
    ll b=readbig();
    if(b == 0){
        cout << "Angry!\n";
    }else{
        ll invb = qpow(b, MOD - 2, MOD);
        ll ans = a * invb % MOD;
        cout << ans << '\n';
    }
    return 0;
}
