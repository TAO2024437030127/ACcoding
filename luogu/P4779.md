给定一个 $n$ 个点，$m$ 条有向边的带非负权图，请你计算从 $s$ 出发，到每个点的距离。

数据保证你能从 $s$ 出发到任意点。

输入格式
第一行为三个正整数 $n, m, s$。
第二行起 $m$ 行，每行三个非负整数 $u_i, v_i, w_i$，表示从 $u_i$ 到 $v_i$ 有一条权值为 $w_i$ 的有向边。

输出格式
输出一行 $n$ 个空格分隔的非负整数，表示 $s$ 到每个点的距离。

$s = 1$；

## 代码

```java
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 100005;
const ll INF = 1e18;

struct Edge{
    int to;
    ll w;
    Edge(int t,ll ww):to(t),w(ww){}
};
vector<Edge> g[MAXN];
int pre[MAXN];//记录最短路径上的前驱
ll dist[MAXN];

void dijkstra(int s,int n){
    fill(dist,dist+n+1,INF);
    fill(pre,pre+n+1,-1);
    priority_queue<pair<ll,int>,vector<pair<ll,int>>,greater<>> pq;
    dist[s]=0;
    pq.emplace(0,s);
    while(!pq.empty()){
        auto [d,u]=pq.top();
        pq.pop();
        if(d>dist[u])continue;//惰性删除？？

        for(const Edge &e:g[u]){
            if(dist[e.to]>dist[u]+e.w){
                dist[e.to]=dist[u]+e.w;
                pre[e.to]=u;
                pq.emplace(dist[e.to],e.to);
            }
        }
    }
}


int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, s;
    cin >> n >> m >> s;
    for (int i = 0; i < m; i++) {
        int u, v; ll w;
        cin >> u >> v >> w;
        g[u].emplace_back(v, w); // 有向；无向则再反向加一次
    }
    dijkstra(s, n);
    for (int i = 1; i <= n; i++)
        cout << (dist[i] == INF ? -1 : dist[i]) << " \n"[i == n];
    return 0;
}
```
