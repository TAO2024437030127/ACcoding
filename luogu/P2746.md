## 题目简述

有一些学校会向其他学校分享软件，即如果这个学校得到了软件，那么在分享列表中的学校也会得到软件。注意这种关系是单向的，即如果 $a$ 在 $b$ 的列表中，那么 $b$ 不一定在 $a$ 的列表中。

现在，你需要向其中一些学校下发新软件。为了节约下发软件的成本，你需要回答以下两个问题。
1. 至少需要向几个学校下发新软件，可以使得所有学校均获得新软件。
2. 定义一次扩展为在某个学校的分享列表中增加一个学校。至少需要进行几次扩展，才可以使得无论对哪个学校**仅下发一次软件**就可以使得所有学校获得新软件。

两个问题相互独立。

输入格式
输入文件的第一行包括一个正整数 $N$，表示学校数目。学校的编号为数字 $1$ 到 $N$。

接下来 $N$ 行，每行都表示一个分享列表，第 $i+1$ 行为学校 $i$ 的分享列表中的学校编号。每个列表用 $0$ 结束，空列表只用一个 $0$ 表示。

输出格式
你的程序应该在输出文件中输出两行。

第一行应该包括一个正整数，表示问题 $1$ 的解。

第二行应该包括一个非负整数，表示问题 $2$ 的解。

说明/提示
$2 \le N \le 100$。

题目翻译来自 NOCOW。

#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN=10000;
vector<int> g[MAXN];
int dfn[MAXN],low[MAXN],stk[MAXN],ins[MAXN],scc[MAXN];
int timer,top,scnt;
void tarjan(int u){
    dfn[u]=low[u]= ++timer;
    stk[++top]=u;
    ins[u]=1;
    for(int v:g[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u]=min(low[u],low[v]);
        }else if(ins[v]){
            low[u]=min(low[u],dfn[v]);
        }
    }
    if(low[u]==dfn[u]){
        scnt++;
        int v;
        do{
            v=stk[top--];
            ins[v]=0;
            scc[v]=scnt;
        }while(v!=u);
    }
}
int main(){
    // 重置所有全局变量
    for(int i=1;i<MAXN;i++) g[i].clear();
    memset(dfn,0,sizeof(dfn));
    memset(low,0,sizeof(low));
    memset(ins,0,sizeof(ins));
    memset(scc,0,sizeof(scc));
    timer = top = scnt = 0;

    int n;
    cin>>n;
    for(int i=1;i<=n;i++){
        int v;
        while(cin>>v&&v!=0){
            g[i].push_back(v);
        }
    }
    for(int i=1;i<=n;i++){
        if(!dfn[i]){
            tarjan(i);
        }
    }
    vector<int> indeg(scnt+1,0),outdeg(scnt+1,0);
    for(int i=1;i<=n;i++){
        for(int v:g[i]){
            if(scc[i]!=scc[v]){
                outdeg[scc[i]]++;
                indeg[scc[v]]++;
            }
        }
    }
    int in0=0, out0=0;
    for(int i=1;i<=scnt;i++)
    {
        if(indeg[i]==0) in0++;
        if(outdeg[i]==0) out0++;
    }
    int ans1 = in0;
    int ans2;
    if(scnt == 1) ans2 = 0;
    else ans2 = max(in0, out0);
    cout << ans1 << endl << ans2 << endl;
    return 0;
}
