## 题目简述

给定一个 $n$ 个点 $m$ 条边有向图，每个点有一个权值，求一条路径，使路径经过的点权值之和最大。你只需要求出这个权值和。

允许多次经过一条边或者一个点，但是，重复经过的点，权值只计算一次。

输入格式
第一行两个正整数 $n,m$。

第二行 $n$ 个整数，其中第 $i$ 个数 $a_i$ 表示点 $i$ 的点权。

第三至 $m+2$ 行，每行两个整数 $u,v$，表示一条 $u\rightarrow v$ 的有向边。

输出格式
共一行，最大的点权之和。

说明/提示
对于 $100\%$ 的数据，$1\le n \le 10^4$，$1\le m \le 10^5$，$0\le a_i\le 10^3$。

#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 10010;

vector<int> g[MAXN];     // 原图
vector<int> dag[MAXN];   // 缩点后的DAG
int n,m;
int a[MAXN];             // 原点点权

int dfn[MAXN], low[MAXN], tim;
int scc[MAXN];           // scc[u]：u属于哪个SCC
int scc_cnt;             // SCC总数
bool in_stk[MAXN];
stack<int> stk;

int val[MAXN];           // 缩点后每个SCC的权值
int deg[MAXN];           // DAG入度
int dp[MAXN];

void tarjan(int u){
    dfn[u] = low[u] = ++tim;
    stk.push(u);
    in_stk[u] = true;
    for(int v : g[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }else if(in_stk[v]){
            low[u] = min(low[u], dfn[v]);
        }
    }
    if(dfn[u] == low[u]){
        ++scc_cnt;
        while(true){
            int now = stk.top();
            stk.pop();
            in_stk[now] = false;
            scc[now] = scc_cnt;
            val[scc_cnt] += a[now];
            if(now == u) break;
        }
    }
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    for(int i=1;i<=n;i++) cin >> a[i];
    for(int i=1;i<=m;i++){
        int u,v; cin >> u >> v;
        g[u].push_back(v);
    }

    // Tarjan求SCC
    for(int i=1;i<=n;i++){
        if(!dfn[i]) tarjan(i);
    }

    // 建DAG，去重边！避免同一个SCC内部连边，也避免重复边
    set<pair<int,int>> edge_set;
    for(int u=1;u<=n;u++){
        for(int v : g[u]){
            int su = scc[u], sv = scc[v];
            if(su != sv && !edge_set.count({su,sv})){
                edge_set.insert({su,sv});
                dag[su].push_back(sv);
                deg[sv]++;
            }
        }
    }

    // 拓扑DP
    queue<int> q;
    for(int i=1;i<=scc_cnt;i++){
        dp[i] = val[i];
        if(deg[i]==0) q.push(i);
    }
    int ans = 0;
    while(!q.empty()){
        int u = q.front(); q.pop();
        ans = max(ans, dp[u]);
        for(int v : dag[u]){
            dp[v] = max(dp[v], dp[u] + val[v]);
            deg[v]--;
            if(deg[v]==0) q.push(v);
        }
    }
    cout << ans << endl;
    return 0;
}
