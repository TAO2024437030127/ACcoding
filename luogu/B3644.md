## 题目简述

有个人的家族很大，辈分关系很混乱，请你帮整理一下这种关系。给出每个人的后代的信息。输出一个序列，使得每个人的后辈都比那个人后列出。

输入格式
第 $1$ 行一个整数 $N$（$1 \le N \le 100$），表示家族的人数。接下来 $N$ 行，第 $i$ 行描述第 $i$ 个人的后代编号 $a_{i,j}$，表示 $a_{i,j}$ 是 $i$ 的后代。每行最后是 $0$ 表示描述完毕。

输出格式
输出一个序列，使得每个人的后辈都比那个人后列出。如果有多种不同的序列，输出任意一种即可。

### 样例 1

输入：

5

5 3 0
3 0

输出：

## 代码

#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN=100005;
vector<int> g[MAXN];
int indeg[MAXN];
int ans[MAXN];
int n,m,tot;

bool toposort(){
    queue<int> q;
    for(int i=1;i<=n;i++){
        if(indeg[i]==0){
            q.push(i);
        }
        
    }
    while(!q.empty()){
            int u=q.front();
            q.pop();
            ans[++tot]=u;
            for(int v:g[u]){
                indeg[v]--;
                if(indeg[v]==0){
                    q.push(v);
                }
                
            }
        }
    return tot==n;
}
void add_edge(int u,int v){
    g[u].push_back(v);
}

int main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        int v;
        while(cin>>v&&v!=0){
            add_edge(i,v);
            indeg[v]++;
        }
    }
    toposort();
    for(int i=1;i<=n;i++){
        cout<<ans[i]<<' ';
    }
    return 0;
}
