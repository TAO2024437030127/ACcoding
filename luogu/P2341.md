## 题目简述

本题测试数据已修复。

每头奶牛都梦想成为牛棚里的明星。被所有奶牛喜欢的奶牛就是一头明星奶牛。所有奶牛都是自恋狂，每头奶牛总是喜欢自己的。奶牛之间的“喜欢”是可以传递的——如果 $A$ 喜欢 $B$，$B$ 喜欢 $C$，那么 $A$ 也喜欢 $C$。牛栏里共有 $N$ 头奶牛，给定一些奶牛之间的爱慕关系，请你算出有多少头奶牛可以当明星。

输入格式
第一行：两个用空格分开的整数：$N$ 和 $M$。

接下来 $M$ 行：每行两个用空格分开的整数：$A$ 和 $B$，表示 $A$ 喜欢 $B$。

输出格式
一行单独一个整数，表示明星奶牛的数量。

说明/提示
只有 $3$ 号奶牛可以做明星。

**【数据范围】**

对于 $10\%$ 的数据，$N\le20$，$M\le50$。

对于 $30\%$ 的数据，$N\le10^3$，$M\le2\times 10^4$。

对于 $70\%$ 的数据，$N\le5\times 10^3$，$M\le5\times 10^4$。

对于 $100\%$ 的数据，$1\le N\le10^4$，$1\le M\le5\times 10^4$。

#include<bits/stdc++.h>
using namespace std;
const int MAXN=100005;
vector<int> g[MAXN];
int dfn[MAXN],low[MAXN],stk[MAXN],ins[MAXN],scc[MAXN];
int timer,top,scnt;
int sz[MAXN];    // 每个SCC大小
int outd[MAXN]; 
void tarjan(int u){
    dfn[u]=low[u]= ++timer;
    stk[++top]=u;
    ins[u]=1;
    for(int v:g[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u]=min(low[u],low[v]);
        }else if(ins[v]){
            low[u]=min(low[u],dfn[v]);
        }
    }
    if(low[u]==dfn[u]){
        scnt++;
        int v;
        do{
            v=stk[top--];
            ins[v]=0;
            scc[v]=scnt;
            sz[scnt]++;
            
        }while(v!=u);
    }
}


int main(){
    int n,m;
    cin>>n>>m;
    for(int i=1;i<=m;i++){
        int u,v;
        cin>>u>>v;
        g[u].push_back(v);
    }
    memset(dfn,0,sizeof(dfn));
    memset(ins,0,sizeof(ins));
    timer=top=scnt=0;
    for(int i=1;i<=n;i++){
        if(!dfn[i]){
            tarjan(i);
        }
    }
    for(int u=1;u<=n;u++){
        for(int v:g[u]){
            int su=scc[u],sv=scc[v];
            if(su!=sv){
                outd[su]++;
            }
        }
    }
    int cnt0=0,ans=0;
    for(int i=1;i<=scnt;i++){
        if(outd[i]==0){
            cnt0++;
            ans=sz[i];
        }
    }
    if(cnt0>1) cout<<0;
    else cout<<ans;
    return 0;
    
    
}
