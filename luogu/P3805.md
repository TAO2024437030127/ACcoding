## 题目简述

给出一个只由小写英文字符 $\texttt a,\texttt b,\texttt c,\ldots\texttt y,\texttt z$ 组成的字符串 $S$ ,求 $S$ 中最长回文串的长度 。

字符串长度为 $n$。

输入格式
一行小写英文字符 $\texttt a,\texttt b,\texttt c,\cdots,\texttt y,\texttt z$ 组成的字符串 $S$。

输出格式
一个整数表示答案。

说明/提示
$1\le n\le 1.1\times 10^7$。

```text
aaa
```

## 代码

```text
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main(){
    string s;
    cin>>s;
    string t;
    t+='#';
    for(auto c:s){
        t+=c;
        t+='#';
    }
    int n=t.size();
    vector<int> p(n,0);
    int mid=0,r=0;
    int ans=0;
    for(int i=0;i<n;i++){
        if(i<r){
            p[i]=min(r-i,p[2*mid-i]);
        }else{
            p[i]=1;
        }
        while(i-p[i]>=0&&i+p[i]<n&&t[i-p[i]]==t[i+p[i]]){
            p[i]++;
        }
        if(i+p[i]>r){
            mid=i;
            r=i+p[i];
        }
         ans=max(ans,p[i]);
    }
    cout<<ans-1<<endl;
    return 0;
    
}
```
