## 题目简述

upd: 新增一组 hack 数据。

农夫约翰想要建造一个围栏用来围住他的奶牛，可是他资金匮乏。他建造的围栏必须包括他的奶牛喜欢吃草的所有地点。对于给出的这些地点的坐标，计算最短的能够围住这些点的围栏的长度。

输入格式
输入数据的第一行是一个整数。表示农夫约翰想要围住的放牧点的数目 $n$。

第 $2$ 到第 $(n + 1)$ 行，每行两个实数，第 $(i + 1)$ 行的实数 $x_i, y_i$ 分别代表第 $i$ 个放牧点的横纵坐标。

输出格式
输出输出一行一个四舍五入保留两位小数的实数，代表围栏的长度。

说明/提示
#### 数据规模与约定

对于 $100\%$ 的数据，保证 $3 \leq n \leq 10^5$，$-10^6 \leq x_i, y_i \leq 10^6$。小数点后最多有 $2$ 位数字。

```text
#include<bits/stdc++.h>
using namespace std;
typedef double db;
const int MAXN=100005;
db px[MAXN],py[MAXN];
int ord[MAXN],stk[MAXN];
int n,top;

db cross3(int o,int a,int b){
    return (px[a]-px[o])*(py[b]-py[o])-(py[a]-py[o])*(px[b]-px[o]);
}
bool cmp_point(int a,int b){
    if(fabs(px[a]-px[b])>1e-8) return px[a]<px[b];
    return py[a]<py[b];
}
void andrew(){
    for(int i=1;i<=n;i++){
        ord[i]=i;
    }
    sort(ord+1,ord+n+1,cmp_point);
    top=0;
    for(int i=1;i<=n;i++){
        while(top>=2&&cross3(stk[top-1],stk[top],ord[i])<=1e-8){
            top--;
        }
        stk[++top]=ord[i];
    }
    int lower=top;
    for(int i=n-1;i>=1;i--){
        // 修正括号！！
        while(top>lower&&cross3(stk[top-1],stk[top],ord[i])<=1e-8){
            top--;
        }
        stk[++top]=ord[i];
    }
    top--;
}
db get_dist(int a,int b){
    db dx=px[a]-px[b];
    db dy=py[a]-py[b];
    return sqrt(dx*dx+dy*dy);
}
int main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>px[i]>>py[i];
    }
    andrew();
    db ans=0;
    for(int i=1;i<=top;i++){
        int j=i%top+1;
        ans += get_dist(stk[i],stk[j]);
    }
    printf("%.2lf\n",ans);
    return 0;
}
```
