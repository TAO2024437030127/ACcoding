## 题目简述

有一个 $n$ 个点，$m$ 条边的有向图，请求出这个图点数大于 $1$ 的强连通分量个数。

输入格式
第一行为两个整数 $n$ 和 $m$。

第二行至 $m+1$ 行，每一行有两个整数 $a$ 和 $b$，表示有一条从 $a$ 到 $b$ 的有向边。

输出格式
仅一行，表示点数大于 $1$ 的强连通分量个数。

说明/提示
#### 数据规模与约定

对于全部的测试点，保证 $2\le n \le 10^4$，$2\le m\le 5\times 10^4$，$1 \leq a, b \leq n$。

#include<bits/stdc++.h>
using namespace std;
const int MAXN=100005;
vector<int> g[MAXN];
int dfn[MAXN],low[MAXN],stk[MAXN],ins[MAXN],scc[MAXN];
int timer,top,scnt;
int sz[MAXN];    // 每个SCC大小
int outd[MAXN]; 
void tarjan(int u){
    dfn[u]=low[u]= ++timer;
    stk[++top]=u;
    ins[u]=1;
    for(int v:g[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u]=min(low[u],low[v]);
        }else if(ins[v]){
            low[u]=min(low[u],dfn[v]);
        }
    }
    if(low[u]==dfn[u]){
        scnt++;
        int v;
        do{
            v=stk[top--];
            ins[v]=0;
            scc[v]=scnt;
            sz[scnt]++;
            
        }while(v!=u);
    }
}

int main(){
    int n,m;
    cin>>n>>m;
    for(int i=1;i<=m;i++){
        int u,v;
        cin>>u>>v;
        g[u].push_back(v);
    }
    memset(dfn,0,sizeof(dfn));
    memset(sz,0,sizeof(sz));
    memset(ins,0,sizeof(ins));
    for(int i=1;i<=n;i++){
        if(!dfn[i]){
            tarjan(i);
        }
    }
    int cnt=0;
    for(int i=1;i<=scnt;i++){
        if(sz[i]>1){
            cnt++;
        }
    }
    cout<<cnt<<endl;
    return 0;
}
