## 题目简述

判断一个点与已知三角形的位置关系。

输入格式
前三行，每行一个坐标，表示该三角形的三个顶点。

第四行，一个点的坐标，试判断该点与前三个点围成三角形的位置关系。

所有坐标值均为整数。

输出格式
- 若点在三角形内（不含边界），输出 $1$；
- 若点在三角形外（不含边界），输出 $2$；
- 若点在三角形边界上（不含顶点），输出 $3$；
- 若点在三角形顶点上，输出 $4$。

说明/提示
### 数据规模与约定

对于 $100\%$ 数据，$0\le x_i,y_i\le 100$。

#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;

// 点结构体
struct Point {
    ll x, y;
    Point() {}
    Point(ll x_, ll y_) : x(x_), y(y_) {}
    // 判断两个点是否完全相同
    bool operator == (const Point &p) const {
        return x == p.x && y == p.y;
    }
};

// 叉积：向量ab × 向量ac
ll cross(Point a, Point b, Point c) {
    return (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
}

// 判断点p是否落在 线段a-b 上（包含两端顶点）
bool onSegment(Point a, Point b, Point p) {
    if (cross(a, b, p) != 0) return false; // 不共线直接false
    // p的坐标必须夹在a,b之间
    if (p.x < min(a.x, b.x) || p.x > max(a.x, b.x)) return false;
    if (p.y < min(a.y, b.y) || p.y > max(a.y, b.y)) return false;
    return true;
}

int main() {
    char ch;
    Point A, B, C, P;
    // 读取带括号格式：(x,y)，自动跳过括号和逗号
    cin >> ch >> A.x >> ch >> A.y >> ch;
    cin >> ch >> B.x >> ch >> B.y >> ch;
    cin >> ch >> C.x >> ch >> C.y >> ch;
    cin >> ch >> P.x >> ch >> P.y >> ch;

    // 【情况4】点就是三角形其中一个顶点
    if (P == A || P == B || P == C) {
        cout << 4 << endl;
        return 0;
    }

    // 【情况3】点在三角形三条边上（已经排除顶点，不含顶点）
    if (onSegment(A, B, P) || onSegment(B, C, P) || onSegment(C, A, P)) {
        cout << 3 << endl;
        return 0;
    }

    // 计算三边对应的叉积
    ll c1 = cross(A, B, P);
    ll c2 = cross(B, C, P);
    ll c3 = cross(C, A, P);

    bool all_pos = (c1 > 0) && (c2 > 0) && (c3 > 0);
    bool all_neg = (c1 < 0) && (c2 < 0) && (c3 < 0);

    if (all_pos || all_neg) {
        // 全部同号，点在三角形内部【情况1】
        cout << 1 << endl;
    } else {
        // 叉积正负混杂，点在外部【情况2】
        cout << 2 << endl;
    }

    return 0;
}
