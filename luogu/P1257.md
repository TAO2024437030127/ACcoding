## 题目简述

给定平面上 $n$ 个点，找出其中的一对点的距离，使得在这 $n$ 个点的所有点对中，该距离为所有点对中最小的。

输入格式
第一行一个整数 $n$，表示点的个数。

接下来 $n$ 行，每行两个整数 $x,y$，表示一个点的行坐标和列坐标。

输出格式
仅一行，一个实数，表示最短距离，四舍五入保留 $4$ 位小数。

说明/提示
#### 数据规模与约定

对于 $100\%$ 的数据，保证 $2 \leq n \leq 10^4$，$0 \leq x, y \leq 10^9$。

```text
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN=100005;
const double INF=1e18;
double px[MAXN],py[MAXN];
int ord[MAXN],tmpn[MAXN];
int n;
double sq(double x){
    return x*x;
}
bool cmp_point(int a,int b){
    if(px[a]!=px[b]){
        return px[a]<px[b];
    }
    return py[a]<py[b];
}
bool cmpy(int a,int b){
    return py[a]<py[b];
}
double dist2(int a,int b){
    double dx=px[a]-px[b];
    double dy=py[a]-py[b];
    return sqrt(dx*dx+dy*dy);
}
double closest(int l,int r){
    if(l>=r) return INF;
    if(r-l==1){
        return dist2(ord[l],ord[r]);
    }
    int mid=(l+r)>>1;
    double midx=px[ord[mid]];
    double d=min(closest(l,mid),closest(mid+1,r));
    int k=0;
    for(int i=l;i<=r;i++){
        // 修复这一行！
        if(fabs(px[ord[i]] - midx) < d){
            tmpn[k++]=ord[i];
        }
    }
    sort(tmpn,tmpn+k,cmpy);
    for(int i=0;i<k;i++){
        for(int j=i+1;j<k&& fabs(py[tmpn[j]] - py[tmpn[i]]) < d;j++){
            d=min(d,dist2(tmpn[i],tmpn[j]));
        }
    }
    return d;
    
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>px[i]>>py[i];
        ord[i]=i;
    }
    sort(ord + 1, ord + n + 1, cmp_point);
    double ans = closest(1, n);
    cout << fixed << setprecision(4) << ans << endl;
    return 0;
}
```
