## 题目简述

给定平面上 $n$ 个点，求凸包直径。

输入格式
第一行一个正整数 $n$。

接下来 $n$ 行，每行两个整数 $x,y$，表示一个点的坐标。保证所有点的坐标两两不同。

输出格式
输出一行一个整数，表示答案的平方。

说明/提示
### 数据范围
对于 $100\%$ 的数据，$2\le n\le 5\times 10^4$，$|x|,|y|\le 10^4$。

```text
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN=100005;
ll px[MAXN],py[MAXN];
int id[MAXN],stk[MAXN],hull[MAXN];
int n,hn;
ll cross_o(int o,int a,int b){
    return (px[a]-px[o])*(py[b]-py[o])-(py[a]-py[o])*(px[b]-px[o]);
}
ll dist2(int a,int b){
    ll dx=px[a]-px[b];
    ll dy=py[a]-py[b];
    return dx*dx+dy*dy;
}

bool cmp_pt(int a,int b){
    if(px[a]!=px[b]){
        return  px[a]<px[b];
    }
    return py[a]<py[b];
}

void convex_hull(){
    for(int i=1;i<=n;i++){
        id[i]=i;
    }
    sort(id+1,id+n+1,cmp_pt);
    int top=0;
    for(int i=1;i<=n;i++){
        while(top>=2&&cross_o(stk[top-1],stk[top],id[i])<=0){
            top--;
        }
        stk[++top]=id[i];
    }
    int lower=top;
    for(int i=n-1;i>=1;i--){
        while(top>lower&&cross_o(stk[top-1],stk[top],id[i])<=0){
            top--;
        }
        stk[++top]=id[i];
    }
    hn=top-1;
    for(int i=1;i<=hn;i++){
        hull[i]=stk[i];
    }
    
}

ll area2(int a,int b,int c){
    ll t=cross_o (a,b,c);
    return t<0?-t:t;
}

ll caliper_diameter(){
    if(hn==1) return 0;
    if(hn==2) return dist2(hull[1],hull[2]);
    ll res=0;
    int j=3;
    for(int i=1;i<=hn;i++){
        int ni=i%hn+1;
        while(1){
            int nj=j%hn+1;
            if(area2(hull[i],hull[ni],hull[nj])>area2(hull[i],hull[ni],hull[j])){
                j=nj;
            }else{
                break;
            }
            
        }
        ll d1=dist2(hull[i],hull[j]);
        ll d2=dist2(hull[ni],hull[j]);
        if(d1>res){
            res=d1;
        }
        if(d2>res){
            res=d2;
        }
    }
    return res;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>px[i]>>py[i];
    }
    convex_hull();
    cout<<caliper_diameter()<<endl;
    return 0;
}
```
